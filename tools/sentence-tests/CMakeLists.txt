cmake_minimum_required(VERSION 3.22)
project(foxlet_sentence_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(SENTENCE_SANITIZERS "Check sentence code with ASan/UBSan" OFF)
if(SENTENCE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(FOXLET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_subdirectory("${FOXLET_ROOT}/native/sentence" sentence)
find_package(Threads REQUIRED)
add_executable(sentence_test sentence_test.cpp core_cases.cpp russian_cases.cpp measurement_cases.cpp)
target_link_libraries(sentence_test PRIVATE foxlet_sentence Threads::Threads)
add_executable(sentence_cli sentence_cli.cpp)
target_link_libraries(sentence_cli PRIVATE foxlet_sentence)
enable_testing()
add_test(NAME sentence_rules COMMAND sentence_test "${CMAKE_CURRENT_SOURCE_DIR}/data/SentenceBreakTest-17.0.0.txt")
